server <- function(input, output) {

# ---------- Display parameters ------- #

    Large_jar_liquid_litres <- 0.8
    Daphnia_algae_C_ad_libitum_mg_per_L_per_day <- 2
    N_days_per_feeding_event <- 2
    extra_food_ratio <- 0.25
    Diet_Microcystis_fraction <- c(0.5, 0, 0, 0)
    Diet_Synechococcus_fraction <- c(0, 0.5, 0, 0)
    Diet_Nannochloropsis_fraction <- c(0, 0, 1, 1)
    Diet_Tetradesmus_fraction <- c(0.5, 0.5, 0, 0)
    Tetradesmus_OD_to_C_slope <- 158.32
    Tetradesmus_OD_to_C_intercept <- -10.991
    Synechococcus_OD_to_C_slope <- 89.419
    Synechococcus_OD_to_C_intercept <- -4.463
    Microcystis_OD_to_C_slope <- 120.15
    Microcystis_OD_to_C_intercept <- -0.41
    Nannochloropsis_OD_to_C_slope <- 96.73
    Nannochloropsis_OD_to_C_intercept <- 0.43

  output$parameters_echo <- renderTable({
    data.frame(
      Parameter = c("Large_jar_liquid_litres", "Daphnia_algae_C_ad_libitum_mg_per_L_per_day",  "N_days_per_feeding_event", "extra_food_ratio", "Diet_Microcystis_fraction",  "Diet_Synechococcus_fraction", "Diet_Nannochloropsis_fraction",  "Diet_Tetradesmus_fraction", "Tetradesmus_OD_to_C_slope", "Tetradesmus_OD_to_C_intercept",  "Synechococcus_OD_to_C_slope", "Synechococcus_OD_to_C_intercept",  "Microcystis_OD_to_C_slope", "Microcystis_OD_to_C_intercept",  "Nannochloropsis_OD_to_C_slope", "Nannochloropsis_OD_to_C_intercept" ),
      Value = c("0.8", "2", "2", "0.25", "c(0.5, 0, 0, 0)", "c(0, 0.5, 0, 0)",  "c(0, 0, 1, 1)", "c(0.5, 0.5, 0, 0)", "158.32", "-10.991", "89.419",  "-4.463", "120.15", "-0.41", "96.73", "0.43")
    )
  })

# ---------- Create a table with diets ------- #

  species_names <- c("Microcystis", "Synechococcus", "Nannochloropsis", "Tetradesmus")
  mix_names <- c("Microcystis + Tetradesmus", "Synechococcus + Tetradesmus", "Nannochloropsis")

  P_diet <- c(0.25, 0.25, 0.50)
  P_alga <- rbind(
    Diet_Microcystis_fraction,
    Diet_Synechococcus_fraction,
    Diet_Nannochloropsis_fraction,
    Diet_Tetradesmus_fraction
  )[, 1:3]
  dimnames(P_alga) <- list(species_names, mix_names)

# ---------- Calculate from OD-carbon relationship the carbon of experimental bottles ------- #

  daphnia_food <- reactive({
    od_by_species <- c(
      input$od_microcystis,
      input$od_synechococcus,
      input$od_nannochloropsis,
      input$od_tetradesmus
    )
    od_to_c_slope <- c(
      Microcystis_OD_to_C_slope,
      Synechococcus_OD_to_C_slope,
      Nannochloropsis_OD_to_C_slope,
      Tetradesmus_OD_to_C_slope
    )
    od_to_c_intercept <- c(
      Microcystis_OD_to_C_intercept,
      Synechococcus_OD_to_C_intercept,
      Nannochloropsis_OD_to_C_intercept,
      Tetradesmus_OD_to_C_intercept
    )
    D_alga <- od_to_c_slope * od_by_species + od_to_c_intercept

# ---------- Calculate other parameters ------- #

    H_alga <- c(if (input$harvest_extra_microcystis) 2 else 1, 1, 1, 1)
    E_food <- 1 + extra_food_ratio

    daily_carbon_demand <- input$n_cultures * Large_jar_liquid_litres * Daphnia_algae_C_ad_libitum_mg_per_L_per_day
    alga_per_carbon <- as.vector(t(P_alga) %*% (1 / D_alga))

    target_conc <- Daphnia_algae_C_ad_libitum_mg_per_L_per_day * N_days_per_feeding_event * E_food

    V_alga <- daily_carbon_demand * N_days_per_feeding_event * E_food * H_alga / D_alga *
      as.vector(P_alga %*% (P_diet))
    V_mix <- daily_carbon_demand * N_days_per_feeding_event * E_food * P_diet * alga_per_carbon

    V_feed_per_jar <- Large_jar_liquid_litres * Daphnia_algae_C_ad_libitum_mg_per_L_per_day * E_food * alga_per_carbon * N_days_per_feeding_event

    V_feed_lysed <- Large_jar_liquid_litres * target_conc * (P_alga[1, 1] / D_alga[1])

    list(
      species_names = species_names,
      mix_names = mix_names,
      V_alga = V_alga,
      V_mix = V_mix,
      V_feed_per_jar = V_feed_per_jar,
      V_feed_lysed = V_feed_lysed
    )
  })

  # ---------- Display the output (in ml) ------- #


  output$outputs_single_echo <- renderTable({
    food <- daphnia_food()
    data.frame(Species = food$species_names, Value = ceiling(food$V_alga * 1000))
  }, digits = 0)

  output$outputs_mix_echo <- renderTable({
    food <- daphnia_food()
    data.frame(Mix = food$mix_names, Value = ceiling(food$V_mix * 1000))
  }, digits = 0)

  output$outputs_jar_echo <- renderTable({
    food <- daphnia_food()
    data.frame(Mix = food$mix_names, Value = ceiling(food$V_feed_per_jar * 1000))
  }, digits = 0)

  output$outputs_lysed_echo <- renderTable({
    food <- daphnia_food()
    data.frame(Species = "Microcystis", Value = ceiling(food$V_feed_lysed * 1000))
  }, digits = 0)
}

shinyApp(ui, server)

Algae to harvest (mL)

\[ V_{\mathrm{alga}} = \frac{ N_{\mathrm{jars}} V_{\mathrm{jar}} F_{\mathrm{feed}} N_{\mathrm{days}} E_{\mathrm{food}} H_{\mathrm{alga}} }{ D_{\mathrm{alga}} } \sum_{d=1}^{N_{\mathrm{diets}}} P_{\mathrm{diet},d} P_{\mathrm{alga},d} \]

Algae to feed each Daphnia jar (mL per feeding event)

\[ V_{\mathrm{feed},d} = \frac{ V_{\mathrm{jar}} F_{\mathrm{feed}} N_{\mathrm{days}} E_{\mathrm{food}} }{ D_a } \]

where:

  • \(V_{\mathrm{alga}}\) = total volume of algal culture required for the experiment [L]
  • \(V_{\mathrm{mix},d}\) = total volume of diet \(d\) to prepare [L]
  • \(V_{\mathrm{feed},d}\) = volume of diet \(d\) added to a single Daphnia jar at each feeding event [L jar\(^{-1}\) feeding event\(^{-1}\)]
  • \(N_{\mathrm{jars}}\) = total number of Daphnia jars [jars]
  • \(V_{\mathrm{jar}}\) = volume of each Daphnia jar [L jar\(^{-1}\)]
  • \(F_{\mathrm{feed}}\) = feeding requirement [mg C L\(^{-1}\) day\(^{-1}\)]
  • \(N_{\mathrm{days}}\) = number of days the prepared food must last [days]
  • \(E_{\mathrm{food}}\) = extra food multiplier [1–∞]
  • \(H_{\mathrm{alga}}\) = harvest multiplier for the alga [1–∞]
  • \(P_{\mathrm{diet},d}\) = proportion of Daphnia jars receiving diet \(d\) [0–1]
  • \(P_{\mathrm{alga},d,alga}\) = proportion of dietary carbon supplied by alga \(a\) in diet \(d\) [0–1]
  • \(D_{\mathrm{alga}}\) = carbon density of the algal culture [mg C L\(^{-1}\)]
  • \(D_a\) = carbon density of alga \(a\) [mg C L\(^{-1}\)]
  • \(N_{\mathrm{diets}}\) = number of diets
  • \(N_{\mathrm{algae}}\) = number of algae included in a diet
  • \(d\) = diet
  • \(a\) = alga
