server <- function(input, output) {

# ---------- Display parameters ------- #

    Daphnia_algae_C_ad_libitum_mg_per_L_per_day <- 1
    V_damselfly_jar_L <- 0.06
    Diet_Microcystis_fraction <- c(0.3, 0, 0, 0)
    Diet_Synechococcus_fraction <- c(0, 0.3, 0, 0)
    Diet_Nannochloropsis_fraction <- c(0, 0, 1, 1)
    Diet_Tetradesmus_fraction <- c(0.7, 0.7, 0, 0)
    Tetradesmus_OD_to_C_slope <- 158.32
    Tetradesmus_OD_to_C_intercept <- -10.991
    Synechococcus_OD_to_C_slope <- 89.419
    Synechococcus_OD_to_C_intercept <- -4.463
    Microcystis_OD_to_C_slope <- 120.15
    Microcystis_OD_to_C_intercept <- -0.41
    Nannochloropsis_OD_to_C_slope <- 96.73
    Nannochloropsis_OD_to_C_intercept <- 0.43

  output$parameters_echo <- renderTable({
    data.frame(
      Parameter = c("Daphnia_algae_C_ad_libitum_mg_per_L_per_day", "V_damselfly_jar_L",  "Diet_Microcystis_fraction", "Diet_Synechococcus_fraction", "Diet_Nannochloropsis_fraction",  "Diet_Tetradesmus_fraction", "Tetradesmus_OD_to_C_slope", "Tetradesmus_OD_to_C_intercept",  "Synechococcus_OD_to_C_slope", "Synechococcus_OD_to_C_intercept",  "Microcystis_OD_to_C_slope", "Microcystis_OD_to_C_intercept",  "Nannochloropsis_OD_to_C_slope", "Nannochloropsis_OD_to_C_intercept" ),
      Value = c("1", "0.06", "c(0.3, 0, 0, 0)", "c(0, 0.3, 0, 0)", "c(0, 0, 1, 1)",  "c(0.7, 0.7, 0, 0)", "158.32", "-10.991", "89.419", "-4.463",  "120.15", "-0.41", "96.73", "0.43")
    )
  })

# ---------- Create a table with diets ------- #

# Define the names of the species and diets

species_names <- c("Microcystis", "Synechococcus", "Nannochloropsis", "Tetradesmus")
diet_names <- c("Microcystis + Tetradesmus", "Synechococcus + Tetradesmus", "Nannochloropsis", "Nannochloropsis")

# Define the proportion of each alga in each diet

P_alga <- rbind(Diet_Microcystis_fraction,
                Diet_Synechococcus_fraction,
                Diet_Nannochloropsis_fraction,
                Diet_Tetradesmus_fraction)[, 1:4]

dimnames(P_alga) <- list(species_names, diet_names)

# ---------- Calculate Daphnia food requirements ---------- #

daphnia_food <- reactive({

  # ----- User inputs: OD of each algal culture ----- #

  od_by_species <- c(
    input$od_microcystis,
    input$od_synechococcus,
    input$od_nannochloropsis,
    input$od_tetradesmus
  )

  # ----- Convert OD to carbon concentration ----- #

  od_to_c_slope <- c(
  Microcystis_OD_to_C_slope,
  Synechococcus_OD_to_C_slope,
  Nannochloropsis_OD_to_C_slope,
  Tetradesmus_OD_to_C_slope)
  
  od_to_c_intercept <- c(
  Microcystis_OD_to_C_intercept,
  Synechococcus_OD_to_C_intercept,
  Nannochloropsis_OD_to_C_intercept,
  Tetradesmus_OD_to_C_intercept)
  
  D_alga <- od_by_species * od_to_c_slope + od_to_c_intercept

  # ----- Define extra food multiplier ----- #

  E_food <- 1 + input$extra_food_percentage / 100

  # ----- Harvest: algae needed ----- #

  V_alga_harvest_L <-
    input$n_buckets *
    input$bucket_liquid_litres *
    Daphnia_algae_C_ad_libitum_mg_per_L_per_day *
    input$N_days_per_feeding_event *
    E_food *
    rowSums(P_alga) / ncol(P_alga) / # This is 1/N diets * proporion of each diet summed
    D_alga

  # ----- Mix: to create Microc + Tetra / Synech + Tetra ----- #

  # Derive the proportion of each algas harvest assigned to each diet

  P_alga_diet <- P_alga / rowSums(P_alga)
  
  # Derive the volume of each alga to add to each diet
  
  V_alga_diet_L <- V_alga_harvest_L * P_alga_diet

  # ----- Add: to each Daphnia bucket fed for two days ----- #

  # Carbon density of each prepared diet

  D_diet <- colSums(V_alga_diet_L * D_alga) / colSums(V_alga_diet_L)
  
  # Volume of each prepared diet to feed per bucket
  
  V_feed_per_bucket_L <-
  input$bucket_liquid_litres *
  Daphnia_algae_C_ad_libitum_mg_per_L_per_day *
  input$N_days_per_feeding_event /
  D_diet

  # ----- Calculate toxin volume for damselfly jars -----
  
  # Target Microcystis concentration (mg C/L)
C_target_microcystis <-
  Diet_Microcystis_fraction[1] *
  Daphnia_algae_C_ad_libitum_mg_per_L_per_day

# Microcystis stock concentration (mg C/L)
C_microcystis_stock <-
  Microcystis_OD_to_C_slope * input$od_microcystis +
  Microcystis_OD_to_C_intercept

# Volume to add (L)
V_toxins_L <-
  input$N_days_per_feeding_event *
  (C_target_microcystis * V_damselfly_jar_L) /
  (C_microcystis_stock - C_target_microcystis)

  # ----- Return results so that you can use them afterwards ----- #

  list(
    species_names = species_names,
    diet_names = diet_names,
    V_alga_harvest_L = V_alga_harvest_L,
    V_alga_diet_L = V_alga_diet_L,
    V_feed_per_bucket_L = V_feed_per_bucket_L,
    V_toxins_L = V_toxins_L
  )
})

  # ---------- Display the output (in ml) ------- #

  output$output_algae_to_harvest <- renderTable({
    food <- daphnia_food()
    data.frame(Species = food$species_names, 
               Value = food$V_alga_harvest_L * 1000)
  }, digits = 2)

  output$output_algae_to_mix_micro <- renderTable({
    food <- daphnia_food()
    data.frame(Mix = food$species_names, 
               Value = food$V_alga_diet_L[,1] * 1000)
  }, digits = 2)

  output$output_algae_to_mix_syne <- renderTable({
    food <- daphnia_food()
    data.frame(Mix = food$species_names, 
               Value = food$V_alga_diet_L[,2] * 1000)
  }, digits = 2)

  output$output_mix_per_bucket_two_days <- renderTable({
    food <- daphnia_food()
    data.frame(Mix = food$diet_names, 
               Value = food$V_feed_per_bucket_L * 1000)
  }, digits = 2)

  output$output_mix_per_bucket_one_day <- renderTable({
    food <- daphnia_food()
    data.frame(Mix = food$diet_names, 
               Value = food$V_feed_per_bucket_L/2 * 1000)
  }, digits = 2)

  output$output_toxins <- renderTable({
  food <- daphnia_food()
  data.frame(Parameter = "Microcystis",
             Value = food$V_toxins_L * 1000)
  }, digits = 3)
}

shinyApp(ui, server)

Harvest: algae needed for diet

\[ V_{\mathrm{alga}} = \frac{ N_{\mathrm{buckets}} V_{\mathrm{bucket}} C_{\mathrm{ad\,libitum}} N_{\mathrm{days}} E_{\mathrm{food}} }{ D_{\mathrm{alga}} } \sum_{diet=1}^{N_{\mathrm{diet}}} \frac{P_{\mathrm{alga},diet}}{N_{\mathrm{diet}}} \]

Mix: to create Microc + Tetra / Synech + Tetra

\[ V_{\mathrm{alga},diet} = V_{\mathrm{alga}} \frac{ P_{\mathrm{alga},diet} }{ \displaystyle \sum_{diet=1}^{N_{\mathrm{diet}}} P_{\mathrm{alga},diet} } \]

Add: to each Daphnia bucket

\[ V_{\mathrm{feed},diet} = \frac{ V_{\mathrm{bucket}} C_{\mathrm{ad\,libitum}} N_{\mathrm{days}} }{ D_{\mathrm{diet}} } \] \[ D_{\mathrm{diet}} = \frac{ \displaystyle\sum_{\mathrm{alga}=1}^{N_{\mathrm{algae}}} V_{\mathrm{alga},\mathrm{diet}}D_{\mathrm{alga}} }{ \displaystyle\sum_{\mathrm{alga}=1}^{N_{\mathrm{algae}}} V_{\mathrm{alga},\mathrm{diet}} } \]

Add: dissolved toxins to each damselfly jar

\[ V_{\text{toxin}} = \frac{C_{\text{dams}} \cdot V_{\text{dams}} \cdot N_{\text{days}}}{C_{\text{toxin}} - C_{\text{dams}}} \]

Parameter definitions

  • \(V_{\mathrm{alga}}\) = total volume of the algal culture to harvest [L]
  • \(V_{\mathrm{alga},diet}\) = volume of the algal culture to add to the mix of the diet [L]
  • \(V_{\mathrm{feed},diet}\) = volume of the diet mix added to a single Daphnia bucket at each feeding event [L bucket\(^{-1}\) feeding event\(^{-1}\)]
  • \(V_{\text{toxin}}\) = volume of the Microcystis culture added to a single damselfly jar at each feeding event [L jar\(^{-1}\) feeding event\(^{-1}\)]
  • \(C_{\text{dams}}\) = target Microcystis carbon concentration in the damselfly jar (Microcystis fraction of the diet × \(C_{\mathrm{ad\,libitum}}\)) [mg C L\(^{-1}\) day\(^{-1}\)]
  • \(V_{\text{dams}}\) = liquid volume of each damselfly jar [L jar\(^{-1}\)]
  • \(C_{\text{toxin}}\) = carbon density of the Microcystis culture [mg C L\(^{-1}\)]
  • \(N_{\mathrm{buckets}}\) = total number of Daphnia buckets [buckets]
  • \(V_{\mathrm{bucket}}\) = liquid volume of each Daphnia bucket [L bucket\(^{-1}\)]
  • \(C_{\mathrm{ad\,libitum}}\) = Daphnia feeding requirement [mg C L\(^{-1}\) day\(^{-1}\)]
  • \(N_{\mathrm{days}}\) = number of days between feeding events [days]
  • \(E_{\mathrm{food}}\) = extra food multiplier [1–∞]
  • \(P_{\mathrm{alga},diet}\) = proportion of the diet’s carbon supplied by the alga [0–1]
  • \(D_{\mathrm{alga}}\) = carbon density of the algal culture [mg C L\(^{-1}\)]
  • \(D_{\mathrm{diet}}\) = carbon density of the diet mix [mg C L\(^{-1}\)]
  • \(N_{\mathrm{diet}}\) = number of diets
  • \(N_{\mathrm{algae}}\) = number of algae
  • \(diet\) = diet index
  • \(alga\) = alga index
